### ② `buildRoutes()` 里、`return [` **之前**，加两个小工具

（这两个是路由要用的：档案 id 归一 + 插件缺席时的档位兜底。**必须放在同一函数作用域里**。）

```js
  /**
   * 档案 id 归一：纯 QQ 号（1~15 位）或 `_n_名字`。
   * ⚠️ **URL 解码后再校验**：中文名 encodeURIComponent 后只剩 %XX，
   *    解码后必须重新验一遍，杜绝 %2e%2e / %2f 穿越。
   */
  const personKeyOf = (raw) => {
    let id = String(raw ?? '');
    try { id = decodeURIComponent(id); } catch { return null; }
    id = id.trim();
    if (!id || id.includes('/') || id.includes('\\') || id.includes('..') || /[\x00-\x1f]/.test(id)) return null;
    if (/^\d{1,15}$/.test(id)) return id;
    if (/^_n_[^/\\]{1,40}$/.test(id)) return id;
    return null;
  };

  /** 好感分 → 档位（**只在插件缺席时**用的本地兜底；插件在时一律用它的 band）。 */
  const favorBandLocal = (favor) => {
    const v = Number.isFinite(Number(favor)) ? Number(favor) : 50;
    if (v >= 85) return '亲近';
    if (v >= 70) return '好感高';
    if (v >= 55) return '稍友好';
    if (v > 40) return '中性';
    if (v > 30) return '稍冷淡';
    if (v > 15) return '反感';
    return '厌恶';
  };
```

### ③ `return [` 数组里加 5 条路由

从作者那套 `routes.js` 原样抄来（`readBody` / `skillRuntimeContext` / `skillManager` / `json` 都是该作用域里已有的名字，**别改名**）：

```js
    // ── 群友（好感度）────────────────────────────────────────────────
    // 插件在时用它的能力（结算/待确认只有插件有），插件不在时退回核心数据层的只读视图 + 直接改分。
    // settle / pending 刻意**不做降级**：门槛/幅度/防刷规则都在插件里，
    // 没插件就明确 501，而不是偷偷用一套简化算法算出不同的结果。
    {
      method: 'GET', pattern: '/api/memory-people',
      handler: async ({ res, json, url }) => {
        const ctx = skillRuntimeContext();
        const states = skillManager.getCapabilityProviders('memory.favor.states', ctx)[0]?.fn?.() || null;
        const wantId = String(url?.searchParams?.get('userId') || '').trim();
        if (wantId) {
          const key = personKeyOf(wantId);
          if (!key) return json(res, 400, { ok: false, error: '档案 id 不合法' });
          const status = skillManager.getCapabilityProviders('memory.favor.status', ctx)[0]?.fn?.({ userId: key }) || null;
          const ledger = skillManager.getCapabilityProviders('memory.favor.ledger', ctx)[0]?.fn?.({ userId: key, limit: 30 }) || null;
          const person = people.readPerson(key) || people.blankPerson(key);
          return json(res, 200, {
            ok: true, favorCore: !!states, person, status, ledger,
            cycles: Array.isArray(person.cycles) ? person.cycles.slice(-20).reverse() : [],
            impressions: Array.isArray(person.impressions) ? person.impressions.slice(-20).reverse() : []
          });
        }
        const list = Array.isArray(states) && states.length
          ? states
          : people.listPeople().map((p) => ({
            userId: p.userId, name: p.name, favor: p.favor, band: favorBandLocal(p.favor),
            pendingCount: Number(p?.pending?.count) || 0, pendingScore: Number(p?.pending?.score) || 0,
            cycles: Number(p.cycleCount) || 0, lastFavorChangeAt: Number(p.lastFavorChangeAt) || 0,
            lastChange: Array.isArray(p.cycles) && p.cycles.length
              ? { at: Number(p.cycles[p.cycles.length - 1].at) || 0, delta: Number(p.cycles[p.cycles.length - 1].delta) || 0, reason: String(p.cycles[p.cycles.length - 1].reason || '') }
              : null
          }));
        list.sort((a, b) => Number(a.favor) - Number(b.favor));
        return json(res, 200, {
          ok: true,
          favorCore: !!states,
          people: list,
          stats: people.peopleStats(),
          audit: readMemoryAudit({ limit: 30, types: ['favor', 'impression', 'pending'] })
        });
      }
    },
    {
      method: 'GET', pattern: /^\/api\/memory-people\/([^/]+)\/favor$/,
      handler: async ({ res, json, match }) => {
        const key = personKeyOf(match[1]);
        if (!key) return json(res, 400, { ok: false, error: '档案 id 不合法' });
        const ctx = skillRuntimeContext();
        const status = skillManager.getCapabilityProviders('memory.favor.status', ctx)[0]?.fn?.({ userId: key }) || null;
        const person = people.readPerson(key) || people.blankPerson(key);
        return json(res, 200, { ok: true, person, status, favorCore: !!status });
      }
    },
    {
      method: 'POST', pattern: /^\/api\/memory-people\/([^/]+)\/favor$/,
      handler: async ({ req, res, json, match }) => {
        const key = personKeyOf(match[1]);
        if (!key) return json(res, 400, { ok: false, error: '档案 id 不合法' });
        const body = await readBody(req).catch(() => ({}));
        const reason = String(body?.reason || '').trim();
        if (!reason) return json(res, 400, { ok: false, error: '必须填写 reason（会写进审计）' });
        const favor = Number(body?.favor);
        if (!Number.isFinite(favor) || favor < 0 || favor > 100) return json(res, 400, { ok: false, error: 'favor 必须是 0~100 的数字' });
        const ctx = skillRuntimeContext();
        const setFn = skillManager.getCapabilityProviders('memory.favor.set', ctx)[0]?.fn;
        if (setFn) {
          const out = setFn({ userId: key, favor, reason });
          return json(res, 200, { ok: true, via: 'favor-core', result: out });
        }
        const before = people.readPerson(key)?.favor ?? 50;
        const next = people.updatePerson(key, { favor }, { reason });
        if (!next) return json(res, 500, { ok: false, error: '写入失败' });
        logMemoryChange({ type: 'favor', source: 'manual_ui', userId: key, from: before, to: next.favor, delta: Number(next.favor) - Number(before), reason });
        return json(res, 200, { ok: true, via: 'core', person: next });
      }
    },
    {
      method: 'POST', pattern: /^\/api\/memory-people\/([^/]+)\/settle$/,
      handler: async ({ req, res, json, match }) => {
        const key = personKeyOf(match[1]);
        if (!key) return json(res, 400, { ok: false, error: '档案 id 不合法' });
        const body = await readBody(req).catch(() => ({}));
        const fn = skillManager.getCapabilityProviders('memory.favor.settle', skillRuntimeContext())[0]?.fn;
        if (!fn) return json(res, 501, { ok: false, error: '需要「好感度内核」插件（favor-core）才能结算：门槛/幅度/防刷规则都在它里面，核心不做简化降级' });
        return json(res, 200, { ok: true, result: fn({ userId: key, force: true, reason: String(body?.reason || '管理页手动结算') }) });
      }
    },
    {
      method: 'POST', pattern: /^\/api\/memory-people\/([^/]+)\/pending$/,
      handler: async ({ req, res, json, match }) => {
        const key = personKeyOf(match[1]);
        if (!key) return json(res, 400, { ok: false, error: '档案 id 不合法' });
        const body = await readBody(req).catch(() => ({}));
        const fn = skillManager.getCapabilityProviders('memory.favor.pending', skillRuntimeContext())[0]?.fn;
        if (!fn) return json(res, 501, { ok: false, error: '需要「好感度内核」插件（favor-core）才能补/清待确认条数' });
        return json(res, 200, {
          ok: true,
          result: fn({
            userId: key,
            set: body?.set === undefined ? null : Number(body.set),
            add: Number(body?.add) || 0,
            clear: body?.clear === true,
            note: String(body?.note || '管理页调整')
          })
        });
      }
    },
```

> ⚠️ 上面 `POST` 分支用 `readBody(req).catch(...)` 是为了少引一个依赖；
> 作者那套里用的是 `bodyOf(req)`（`routes.js:77` 定义）。**你那边有 `bodyOf` 就换成它**，别两套混用。
